"""Realm (auth) server connection: logon, SRP proof and realm list.

Port of the Client/Authentication/* files. The C# version used asynchronous callbacks
that re-armed a one-byte read; here the whole handshake is a straight-line blocking
sequence, which is equivalent for this strictly request/response protocol.

from __future__ import annotations

import hmac
import socket
import struct
from enum import IntEnum
from typing import TYPE_CHECKING, Iterator, Optional

from .definitions import LogLevel
from .srp import compute_client_proof
from .util import ByteReader, recv_exact, to_cstring, to_hex_string

if TYPE_CHECKING:  # pragma: no cover
    from .game import Game

CLIENT_BUILD = 12340
CLIENT_VERSION = (3, 3, 5)


class AuthCommand(IntEnum):
    LOGON_CHALLENGE = 0x00
    LOGON_PROOF = 0x01
    REALM_LIST = 0x10
    TRANSFER_INITIATE = 0x30
    TRANSFER_DATA = 0x31
    TRANSFER_ACCEPT = 0x32
    TRANSFER_RESUME = 0x33
    TRANSFER_CANCEL = 0x34


class AuthResult(IntEnum):
    SUCCESS = 0
    FAILURE = 0x01
    UNKNOWN1 = 0x02
    ACCOUNT_BANNED = 0x03
    NO_MATCH = 0x04
    UNKNOWN2 = 0x05
    ACCOUNT_IN_USE = 0x06
    PREPAID_TIME_LIMIT = 0x07
    SERVER_FULL = 0x08
    WRONG_BUILD_NUMBER = 0x09
    UPDATE_CLIENT = 0x0A
    UNKNOWN3 = 0x0B
    ACCOUNT_FREEZED = 0x0C
    UNKNOWN4 = 0x0D
    UNKNOWN5 = 0x0E
    PARENTAL_CONTROL = 0x0F


class WorldServerInfo:
    """One entry of the realm list."""

    def __init__(self, reader: ByteReader) -> None:
        self.type = reader.read_u8()
        self.locked = reader.read_u8()
        self.flags = reader.read_u8()
        self.name = reader.read_cstring()
        address = reader.read_cstring()
        host, _, port = address.partition(":")
        self.address = host
        self.port = int(port) if port else 8085
        self.population = reader.read_float()
        self.load = reader.read_u8()
        self.timezone = reader.read_u8()
        self.realm_id = reader.read_u8()  # "unk1" in the C# source
        if self.flags & 4:
            self.version_major = reader.read_u8()
            self.version_minor = reader.read_u8()
            self.version_bugfix = reader.read_u8()
            self.build = reader.read_u16()


class WorldServerList:
    def __init__(self, reader: ByteReader) -> None:
        reader.read_u32()
        count = reader.read_u16()
        self._servers = [WorldServerInfo(reader) for _ in range(count)]

    @property
    def count(self) -> int:
        return len(self._servers)

    def __len__(self) -> int:
        return len(self._servers)

    def __getitem__(self, index: int) -> WorldServerInfo:
        return self._servers[index]

    def __iter__(self) -> Iterator[WorldServerInfo]:
        return iter(self._servers)


class AuthSocket:
    def __init__(self, game: "Game", hostname: str, port: int, username: str, password: str) -> None:
        self.game = game
        self.hostname = hostname
        self.port = port
        self.username = username
        self.password = password
        self.key: bytes = b""  # 40-byte session key after a successful logon
        self._sock: Optional[socket.socket] = None
        self._closing = False

    # -- helpers ---------------------------------------------------------------------

    @property
    def _ui(self):
        return self.game.ui

    def _read_command(self) -> int:
        value = recv_exact(self._sock, 1)[0]
        try:
            return AuthCommand(value)
        except ValueError:
            return value

    def close(self) -> None:
        self._closing = True
        sock = self._sock
        if sock is not None:
            try:
                sock.shutdown(socket.SHUT_RDWR)
            except OSError:
                pass
            sock.close()

    # -- logon -------------------------------------------------------------------------

    def connect(self) -> bool:
        if self._closing:
            return False
        self._ui.log("Connecting to realmlist... ")
        try:
            self._sock = socket.create_connection((self.hostname, self.port), timeout=15)
            if self._closing:
                self.close()
                return False
        except OSError as ex:
            self._ui.log_line(f"failed. ({ex})", LogLevel.Error)
            return False
        self._ui.log_line("done!")
        return True

    def authenticate(self) -> Optional[WorldServerList]:
        """Run the whole realm-server handshake. Returns the realm list, or None on failure."""
        if not self.connect():
            return None
        try:
            return self._authenticate()
        except (ConnectionError, EOFError):
            self._ui.log_line("Server has disconnected.", LogLevel.Info)
            return None
        except OSError as ex:
            self._ui.log_line(f"Network error: {ex}", LogLevel.Error)
            return None
        except ValueError as ex:
            self._ui.log_line(f"Authentication failed: {ex}", LogLevel.Error)
            return None

    def _send_logon_challenge(self) -> None:
        self._ui.log_line("Sending logon challenge", LogLevel.Debug)
        name = self.username.upper().encode("utf-8")
        try:
            ip = socket.inet_aton(self._sock.getsockname()[0])
        except OSError:  # e.g. IPv6 local address
            ip = bytes(4)
        packet = (
            bytes([AuthCommand.LOGON_CHALLENGE, 6])      # command, protocol version
            + struct.pack("<H", len(name) + 30)          # size of the rest
            + to_cstring("WoW")
            + bytes(CLIENT_VERSION)
            + struct.pack("<H", CLIENT_BUILD)
            + to_cstring("68x")      # "x86" reversed
            + to_cstring("niW")      # "Win" reversed
            + b"SUne"                # "enUS" reversed
            + struct.pack("<I", 0x3C)  # timezone bias
            + ip
            + bytes([len(name)])
            + name
        )
        self._sock.sendall(packet)

    def _authenticate(self) -> Optional[WorldServerList]:
        ui = self._ui
        self._send_logon_challenge()

        # ---- server challenge ----
        command = self._read_command()
        if command != AuthCommand.LOGON_CHALLENGE:
            ui.log_line(f"Unknown or unhandled command '{command}'", LogLevel.Debug)
            return None
        reader_head = recv_exact(self._sock, 2)  # unk2, error
        error = reader_head[1]
        if error != AuthResult.SUCCESS:
            messages = {
                AuthResult.NO_MATCH: "Unknown account name",
                AuthResult.ACCOUNT_IN_USE: "Account already logged in",
                AuthResult.WRONG_BUILD_NUMBER: "Wrong build number",
                AuthResult.ACCOUNT_BANNED: "Account is banned",
            }
            ui.log_line(messages.get(error, f"Logon challenge failed (error {error})"), LogLevel.Error)
            return None

        body = ByteReader(recv_exact(self._sock, 32 + 1 + 1 + 1 + 32 + 32 + 16 + 1))
        B = body.read_bytes(32)
        body.read_u8()                 # g length
        g = body.read_bytes(1)
        body.read_u8()                 # N length
        N = body.read_bytes(32)
        salt = body.read_bytes(32)
        body.read_bytes(16)            # version challenge ("unk3")
        security_flags = body.read_u8()
        if security_flags:
            ui.log_line(
                "This account requires extra security (PIN/matrix/authenticator), which is not supported.",
                LogLevel.Error,
            )
            return None
        ui.log_line("Received logon challenge", LogLevel.Debug)

        srp = compute_client_proof(self.username, self.password, B, g, N, salt)
        self.key = srp.K

        ui.log_line("---====== Received from server: ======---", LogLevel.Debug)
        ui.log_line(f"B={to_hex_string(B)}", LogLevel.Debug)
        ui.log_line(f"N={to_hex_string(N)}", LogLevel.Debug)
        ui.log_line(f"salt={to_hex_string(salt)}", LogLevel.Debug)
        ui.log_line("---====== Send data to server: ======---", LogLevel.Debug)
        ui.log_line(f"A={to_hex_string(srp.A)}", LogLevel.Debug)
        ui.log_line(f"S={to_hex_string(srp.S)}", LogLevel.Debug)
        ui.log_line(f"K={to_hex_string(srp.K)}", LogLevel.Debug)
        ui.log_line(f"M={to_hex_string(srp.M1)}", LogLevel.Debug)

        ui.log_line("Sending logon proof", LogLevel.Debug)
        self._sock.sendall(
            bytes([AuthCommand.LOGON_PROOF]) + srp.A + srp.M1 + bytes(20) + b"\x00\x00"
        )

        # ---- server proof ----
        command = self._read_command()
        if command != AuthCommand.LOGON_PROOF:
            ui.log_line(f"Unknown or unhandled command '{command}'", LogLevel.Debug)
            return None
        error = recv_exact(self._sock, 1)[0]
        if error != AuthResult.SUCCESS:
            recv_exact(self._sock, 2)
            if error == AuthResult.UPDATE_CLIENT:
                ui.log_line("Client update requested")
            elif error in (AuthResult.NO_MATCH, AuthResult.UNKNOWN2):
                ui.log_line("Wrong password or invalid account or authentication error", LogLevel.Error)
            elif error == AuthResult.WRONG_BUILD_NUMBER:
                ui.log_line("Wrong build number", LogLevel.Error)
            else:
                ui.log_line(f"Unknown error {error}", LogLevel.Error)
            return None

        proof = ByteReader(recv_exact(self._sock, 20 + 4 + 4 + 2))
        m2 = proof.read_bytes(20)
        ui.log_line("Received logon proof", LogLevel.Debug)
        if not hmac.compare_digest(m2, srp.M2_expected):
            ui.log_line("Server auth failed!", LogLevel.Error)
            return None
        ui.log_line("Authentication succeeded!")

        # ---- realm list ----
        ui.log_line("Requesting realm list", LogLevel.Detail)
        self._sock.sendall(bytes([AuthCommand.REALM_LIST]) + struct.pack("<I", 0))
        command = self._read_command()
        if command != AuthCommand.REALM_LIST:
            ui.log_line(f"Unknown or unhandled command '{command}'", LogLevel.Debug)
            return None
        (size,) = struct.unpack("<H", recv_exact(self._sock, 2))
        realms = WorldServerList(ByteReader(recv_exact(self._sock, size)))
        ui.log_line("Received realm list", LogLevel.Debug)
        return realms

class _Terminal:
    """Single-key input with a timeout, on POSIX ttys, Windows consoles and plain pipes."""

    def __init__(self) -> None:
        self._tty = sys.stdin.isatty()
        self._saved = None
        self._lock = threading.Lock()

    def enter_cbreak(self) -> None:
        if os.name == "nt" or not self._tty:
            return
        import termios
        import tty
        with self._lock:
            if self._saved is None:
                fd = sys.stdin.fileno()
                self._saved = termios.tcgetattr(fd)
                tty.setcbreak(fd)

    def restore(self) -> None:
        if os.name == "nt" or self._saved is None:
            return
        import termios
        with self._lock:
            termios.tcsetattr(sys.stdin.fileno(), termios.TCSADRAIN, self._saved)
            self._saved = None

    def read_key(self, timeout: Optional[float]) -> Optional[str]:
        if not self._tty:
            # Piped input: one line per "key press" (first character), blocking.
            line = sys.stdin.readline()
            if line == "":
                raise EOFError
            return line[0]
        if os.name == "nt":
            import msvcrt
            end = None if timeout is None else time.monotonic() + timeout
            while True:
                if msvcrt.kbhit():
                    return msvcrt.getwch()
                if end is not None and time.monotonic() >= end:
                    return None
                time.sleep(0.01)
        import select
        self.enter_cbreak()
        fd = sys.stdin.fileno()
        ready, _, _ = select.select([fd], [], [], timeout)
        if not ready:
            return None
        data = os.read(fd, 1)
        if not data:
            raise EOFError
        return data.decode("utf-8", errors="ignore")


class CommandLineUI(GameUI):
    def __init__(self) -> None:

        self._lock = threading.RLock()
        self._term = _Terminal()
        self._color = sys.stdout.isatty() and not os.environ.get("NO_COLOR")
        if os.name == "nt":
            os.system("")  # enables ANSI escape processing on Windows 10+ consoles
        self._log_file = open(
            datetime.now().strftime("%Y-%m-%d %H-%M-%S") + ".log", "w", encoding="utf-8", buffering=1
        )
        self._keybinds: Dict[str, Callable[[], None]] = {
            "s": self.do_say_chat,
            "y": self.do_yell_chat,

    def update(self) -> None:

            time.sleep(0.1)
            return
        try:
            key = self._term.read_key(0.1)
            handler = self._keybinds.get(key.lower()) if key else None
            if handler:
                handler()
        except (EOFError, KeyboardInterrupt):
            self.game.exit()

"""Crypto primitives (port of Client/Crypto/* and World/Network/AuthenticationCrypto.cs)."""
from __future__ import annotations

import hashlib
import hmac
import threading
from enum import Enum


def sha1(*parts: bytes) -> bytes:
    """SHA-1 over the concatenation of all parts (HashHelper.Hash)."""
    h = hashlib.sha1()
    for part in parts:
        h.update(part)
    return h.digest()


class ARC4:
    """Plain RC4 stream cipher (the same stream is used for both directions of a transform)."""

    def __init__(self, key: bytes) -> None:
        state = list(range(256))
        j = 0
        for i in range(256):
            j = (j + state[i] + key[i % len(key)]) & 0xFF
            state[i], state[j] = state[j], state[i]
        self._state = state
        self._x = 0
        self._y = 0

    def process(self, data: bytes) -> bytes:
        state = self._state
        x, y = self._x, self._y
        out = bytearray(len(data))
        for n, byte in enumerate(data):
            x = (x + 1) & 0xFF
            y = (state[x] + y) & 0xFF
            state[x], state[y] = state[y], state[x]
            out[n] = byte ^ state[(state[x] + state[y]) & 0xFF]
        self._x, self._y = x, y
        return bytes(out)


class AuthStatus(Enum):
    Uninitialized = 0
    Pending = 1
    Ready = 2


# Fixed HMAC keys of the 3.3.5a world protocol (client -> server / server -> client).
ENCRYPTION_KEY = bytes([
    0xC2, 0xB3, 0x72, 0x3C, 0xC6, 0xAE, 0xD9, 0xB5,
    0x34, 0x3C, 0x53, 0xEE, 0x2F, 0x43, 0x67, 0xCE,
])
DECRYPTION_KEY = bytes([
    0xCC, 0x98, 0xAE, 0x04, 0xE8, 0x97, 0xEA, 0xCA,
    0x12, 0xDD, 0xC0, 0x93, 0x42, 0x91, 0x53, 0x57,
])


class AuthenticationCrypto:
    """Header encryption for one world connection.

    The C# version kept this in static state; here it is an instance owned by the
    WorldSocket. Both directions are no-ops until `initialize` has been called.
    """

    def __init__(self) -> None:
        self.status = AuthStatus.Uninitialized
        self._encrypt_stream: ARC4 | None = None
        self._decrypt_stream: ARC4 | None = None
        self._lock = threading.Lock()

    def initialize(self, session_key: bytes) -> None:
        enc = ARC4(hmac.new(ENCRYPTION_KEY, session_key, hashlib.sha1).digest())
        enc.process(bytes(1024))  # RC4-drop[1024]
        dec = ARC4(hmac.new(DECRYPTION_KEY, session_key, hashlib.sha1).digest())
        dec.process(bytes(1024))
        with self._lock:
            self._encrypt_stream, self._decrypt_stream = enc, dec
            self.status = AuthStatus.Ready

    def encrypt(self, data: bytes) -> bytes:
        if self.status != AuthStatus.Ready:
            return data
        return self._encrypt_stream.process(data)

    def decrypt(self, data: bytes) -> bytes:
        if self.status != AuthStatus.Ready:
            return data
        return self._decrypt_stream.process(data)

class Game:
    def __init__(
        self,
        ui_class: Callable[[], GameUI],
        hostname: str,
        port: int,
        username: str,
        password: str,
        log_level: LogLevel,
    ) -> None:
        self.ui = ui_class()
        self.ui.game = self
        self.ui.log_level = log_level

        self.world = GameWorld()
        self.username = username
        self.key: bytes = b""  # 40-byte session key from the realm server

        self.running = False
        self.interrupted = False
        self._cancelled = Event()
        self._auth = AuthSocket(self, hostname, port, username, password)
        self._world_socket: Optional[WorldSocket] = None

    def connect_to(self, server: WorldServerInfo) -> None:
        if self._cancelled.is_set():
            return
        self.key = self._auth.key
        self._auth.close()

        self._world_socket = WorldSocket(self, server)
        if not self._cancelled.is_set() and self._world_socket.connect() and not self._cancelled.is_set():
            self._world_socket.start()
        else:
            self.exit()

    def start(self) -> None:
        try:
            try:
                realms = None if self._cancelled.is_set() else self._auth.authenticate()
            finally:
                self._auth.password = ""
            if self._cancelled.is_set():
                return
            self.running = realms is not None
            if realms is not None:
                if len(realms) == 0:
                    self.ui.log_line("The realm list is empty.", LogLevel.Error)
                    self.running = False
                else:
                    self.ui.present_realm_list(realms)
            while self.running and not self._cancelled.is_set():
                self.ui.update()
        except KeyboardInterrupt:
            self.interrupted = True
        finally:
            self.running = False
            if self._world_socket is not None:
                self._world_socket.close()
            self._auth.close()
            self.ui.exit()

    def exit(self) -> None:
        self.running = False

    def cancel(self) -> None:
        """Request exit and unblock any socket reads without waiting for a thread."""
        self._cancelled.set()
        self.exit()
        self._auth.close()
        if self._world_socket is not None:
            self._world_socket.close()

    def send_packet(self, packet: OutPacket) -> None:
        if self._world_socket is not None:
            self._world_socket.send(packet)

"""World packet containers (port of World/Network/InPacket.cs, OutPacket.cs, ClientHeader.cs)."""
from __future__ import annotations

import struct

from .crypto import AuthenticationCrypto
from .opcodes import WorldCommand
from .util import ByteReader, to_cstring


class InPacket(ByteReader):
    """A received world packet: opcode plus payload reader."""

    def __init__(self, command: int, data: bytes = b"") -> None:
        super().__init__(data)
        self.command = command


class OutPacket:
    """A world packet being built for sending."""

    def __init__(self, command: WorldCommand) -> None:
        self.command = command
        self._buf = bytearray()

    def write_u8(self, value: int) -> None:
        self._buf += struct.pack("<B", value)

    def write_u16(self, value: int) -> None:

"""Client side of the WoW SRP-6 logon (the maths from Authentication/Network/AuthSocket.cs).

All big numbers on the wire are little-endian and fixed width (32 bytes for the
3.3.5a modulus). The C# version used BigInteger.ToByteArray(), which drops leading
zero bytes (1 in 256 values) and whose ModPow can return a negative result for a
negative base; Python's integers and fixed-width encoding avoid both problems.
"""
from __future__ import annotations

import secrets
from dataclasses import dataclass
from typing import Callable

from .crypto import sha1

K_MULTIPLIER = 3


@dataclass
class SrpResult:
    A: bytes             # client public key (sent to the server)
    M1: bytes            # client proof (sent to the server)
    K: bytes             # 40-byte session key
    M2_expected: bytes   # proof the server must send back
    S: bytes             # raw shared secret (for debug output)


def _le(data: bytes) -> int:
    return int.from_bytes(data, "little")


def password_hash(username: str, password: str) -> bytes:
    return sha1(f"{username}:{password}".upper().encode("utf-8"))


def interleave_session_key(s: bytes) -> bytes:
    """Split S into even/odd bytes, hash each half with SHA-1 and interleave -> 40 bytes."""
    if len(s) != 32:
        raise ValueError("S must be 32 bytes")
    even = sha1(bytes(s[i * 2] for i in range(16)))
    odd = sha1(bytes(s[i * 2 + 1] for i in range(16)))
    key = bytearray(40)
    key[0::2] = even
    key[1::2] = odd
    return bytes(key)


def compute_client_proof(
    username: str,
    password: str,
    B: bytes,
    g: bytes,
    N: bytes,
    salt: bytes,
    rand: Callable[[int], bytes] = secrets.token_bytes,
) -> SrpResult:
    width = len(N)
    n, gi, b = _le(N), _le(g), _le(B)
    if b % n == 0:
        raise ValueError("server sent an invalid public key")

    user = username.upper().encode("utf-8")
    x = _le(sha1(salt, password_hash(username, password)))

    while True:
        a = _le(rand(19))
        A = pow(gi, a, n)
        if A % n != 0:
            break
    a_bytes = A.to_bytes(width, "little")

    u = _le(sha1(a_bytes, B))
    if u == 0:
        raise ValueError("invalid scrambling parameter")

    # S = (B - k*g^x) ^ (a + u*x) mod N   -- reduce the base first so it is never negative
    base = (b - K_MULTIPLIER * pow(gi, x, n)) % n
    s_bytes = pow(base, a + u * x, n).to_bytes(width, "little")
    session_key = interleave_session_key(s_bytes)

    g_n_hash = bytes(x1 ^ x2 for x1, x2 in zip(sha1(N), sha1(g)))
    m1 = sha1(g_n_hash, sha1(user), salt, a_bytes, B, session_key)
    m2 = sha1(a_bytes, m1, session_key)
    return SrpResult(A=a_bytes, M1=m1, K=session_key, M2_expected=m2, S=s_bytes)

def to_hex_string(data: bytes) -> str:
    """Hex string with the most significant byte first (input is little-endian)."""
    return bytes(reversed(data)).hex().upper()


def to_cstring(text: str) -> bytes:
    """NUL-terminated UTF-8 string."""
    return text.encode("utf-8") + b"\x00"


def recv_exact(sock: socket.socket, count: int) -> bytes:
    """Read exactly `count` bytes or raise ConnectionError if the peer closes."""
    chunks = []
    remaining = count
    while remaining:
        chunk = sock.recv(remaining)
        if not chunk:
            raise ConnectionError("connection closed by peer")
        chunks.append(chunk)
        remaining -= len(chunk)
    return b"".join(chunks)


class ByteReader:
    """Little-endian reader over a bytes buffer (replacement for BinaryReader)."""

    def __init__(self, data: bytes = b"") -> None:
        self._data = bytes(data)
        self._pos = 0

    @property
    def remaining(self) -> int:
        return len(self._data) - self._pos

    def _unpack(self, fmt: str):
        size = struct.calcsize(fmt)
        if self._pos + size > len(self._data):
            raise EOFError("unexpected end of data")
        (value,) = struct.unpack_from(fmt, self._data, self._pos)
        self._pos += size
        return value

    def read_u8(self) -> int:
        return self._unpack("<B")

    def read_u16(self) -> int:
        return self._unpack("<H")

    def read_u32(self) -> int:
        return self._unpack("<I")

    def read_i32(self) -> int:
        return self._unpack("<i")

    def read_u64(self) -> int:
        return self._unpack("<Q")

    def read_float(self) -> float:
        return self._unpack("<f")

    def read_bool(self) -> bool:
        return self.read_u8() != 0

    def read_bytes(self, count: int) -> bytes:
        if self._pos + count > len(self._data):
            raise EOFError("unexpected end of data")
        value = self._data[self._pos:self._pos + count]
        self._pos += count
        return value

    def read_cstring(self) -> str:
        end = self._data.find(b"\x00", self._pos)
        if end < 0:
            raise EOFError("unterminated string")
        value = self._data[self._pos:end]
        self._pos = end + 1
        return value.decode("utf-8", errors="replace")

    def read_packed_guid(self) -> int:
        mask = self.read_u8()
        guid = 0
        for i in range(8):
            if mask & (1 << i):
                guid |= self.read_u8() << (i * 8)
        return guid

class WorldSocket:
    def __init__(self, game: "Game", server_info: WorldServerInfo) -> None:
        self.game = game
        self.server_info = server_info
        self.crypto = AuthenticationCrypto()
        self.transferred = 0
        self.sent = 0
        self.received = 0

        self._sock: Optional[socket.socket] = None
        self._send_lock = threading.Lock()
        self._closing = False
        self._queue: "queue.Queue[Optional[tuple]]" = queue.Queue()
        self._handlers: Dict[int, Callable[[InPacket], None]] = {}
        self.init_handlers()

    @property
    def _ui(self):
        return self.game.ui

    # -- handler registration ---------------------------------------------------------

    def init_handlers(self) -> None:
        self._handlers = {
            WorldCommand.ServerAuthChallenge: self.handle_server_auth_challenge,
            WorldCommand.ServerAuthResponse: self.handle_server_auth_response,
            WorldCommand.ServerCharacterEnumeration: self.handle_char_enum,
            WorldCommand.SMSG_LOGIN_VERIFY_WORLD: lambda _packet: self._ui.world_entered(),
            WorldCommand.SMSG_CHARACTER_LOGIN_FAILED: self.handle_character_login_failed,
            WorldCommand.SMSG_MESSAGECHAT: self.handle_message_chat,
            WorldCommand.SMSG_CHAT_PLAYER_NOT_FOUND: self.handle_chat_player_not_found,
            WorldCommand.SMSG_NAME_QUERY_RESPONSE: self.handle_name_query_response,
        }
        for name, handler in self._handlers.items():
            self._ui.log_line(f"Registered '{handler.__name__}' to '{WorldCommand(name).name}'", LogLevel.Debug)

    # -- connection -------------------------------------------------------------------

    def connect(self) -> bool:
        if self._closing:
            return False
        info = self.server_info
        self._ui.log(f"Connecting to realm {info.name}... ")
        try:
            self._sock = socket.create_connection((info.address, info.port), timeout=15)
            if self._closing:
                self.close()
                return False
            self._sock.settimeout(None)
        except OSError as ex:
            self._ui.log_line(f"failed. ({ex})", LogLevel.Error)
            return False
        self._ui.log_line("done!")
        return True

    def start(self) -> None:
        threading.Thread(target=self._worker_loop, name="world-handlers", daemon=True).start()
        threading.Thread(target=self._read_loop, name="world-reader", daemon=True).start()

    def close(self) -> None:
        self._closing = True
        self._queue.put(None)
        if self._sock is not None:
            try:
                self._sock.shutdown(socket.SHUT_RDWR)
            except OSError:
                pass
            self._sock.close()

    # -- sending ----------------------------------------------------------------------

    def send(self, packet: OutPacket) -> None:
        # Encrypt and send under one lock: RC4 is a stream, so order on the wire must
        # equal order of encryption (the C# version encrypted outside its lock).
        with self._send_lock:
            data = packet.build(self.crypto)
            self._sock.sendall(data)
        self.transferred += len(data)
        self.sent += len(data)

    # -- receiving --------------------------------------------------------------------

    def _read_header(self):
        """Server header: 2-byte big-endian size (3 bytes if the top bit of the first
        byte is set), then a 2-byte little-endian opcode. Size includes the opcode."""
        first = self.crypto.decrypt(recv_exact(self._sock, 1))[0]
        if first & 0x80:
            rest = self.crypto.decrypt(recv_exact(self._sock, 4))
            size = ((first & 0x7F) << 16) | (rest[0] << 8) | rest[1]
            (command,) = struct.unpack_from("<H", rest, 2)
            consumed = 5
        else:
            rest = self.crypto.decrypt(recv_exact(self._sock, 3))
            size = (first << 8) | rest[0]
            (command,) = struct.unpack_from("<H", rest, 1)
            consumed = 4
        self.transferred += consumed
        self.received += consumed
        return command, size - 2

    def _read_loop(self) -> None:
        try:
            while True:
                command, length = self._read_header()
                payload = recv_exact(self._sock, length) if length > 0 else b""
                self.transferred += length
                self.received += length
                self._ui.log_line(f"Command {command} Packet Size {length}", LogLevel.Debug)
                self._dispatch(InPacket(command, payload))
        except (ConnectionError, OSError, EOFError):
            if not self._closing:
                self._ui.log_line("Server has closed the connection")
                self.game.exit()

    def _dispatch(self, packet: InPacket) -> None:
        handler = self._handlers.get(packet.command)
        if handler is None:
            if packet.command not in IGNORED_COMMANDS:
                self._ui.log_line(f"Unknown or unhandled command '{packet.command}'", LogLevel.Debug)
            return
        self._ui.log_line(f"Received {packet.command}", LogLevel.Debug)
        if self.crypto.status == AuthStatus.Ready:
            self._queue.put((handler, packet))  # handled asynchronously, in order
        else:
            handler(packet)  # the auth challenge must be handled before the next read

    def _worker_loop(self) -> None:
        while True:
            item = self._queue.get()
            if item is None:
                return
            handler, packet = item
            try:
                handler(packet)
            except Exception as ex:  # keep the client alive if one packet is malformed
                self._ui.log_line(f"Error handling command {packet.command}: {ex!r}", LogLevel.Error)
                self._ui.log_line(traceback.format_exc(), LogLevel.Debug)

def parse_toc_version(content):
    match = re.search(r"^##\s*Version:\s*(.+)$", content, re.MULTILINE | re.IGNORECASE)
    return match.group(1).strip() if match else "N/A"


def get_local_version(wow_dir, addon_name):
    addon_path = os.path.join(wow_dir, "Interface", "AddOns", addon_name)
    if not os.path.isdir(addon_path):
        return None

    toc_file = os.path.join(addon_path, f"{addon_name}.toc")
    if not os.path.exists(toc_file):
        for fname in os.listdir(addon_path):
            if fname.lower().endswith(".toc"):
                toc_file = os.path.join(addon_path, fname)
                break

    if os.path.exists(toc_file):
        try:
            with open(toc_file, "r", encoding="utf-8", errors="ignore") as f:
                return parse_toc_version(f.read())
        except Exception:
            return "?"
    return "Instalado"


def group_addon_folders(folder_names):
    folder_set = set(folder_names)
    groups = {}

    for name in sorted(folder_names, key=lambda s: (len(s), s.lower())):
        if "_" in name:
            prefix = name.split("_", 1)[0]
            if prefix in folder_set:
                groups.setdefault(prefix, []).append(name)
                continue
        groups.setdefault(name, [])
        if name not in groups[name]:
            groups[name].insert(0, name)

    return groups


def scan_local_addons(wow_dir):
    addons_dir = os.path.join(wow_dir, "Interface", "AddOns")
    if not os.path.isdir(addons_dir):

    raw_folders = [
        name for name in os.listdir(addons_dir)
        if not name.startswith("Blizzard_") and not name.startswith(".") and os.path.isdir(os.path.join(addons_dir, name))
    ]
    groups = group_addon_folders(raw_folders)

    local_list = []
    for main_name in sorted(groups.keys(), key=str.lower):
        ver = get_local_version(wow_dir, main_name)
        local_list.append((main_name, ver, "Local", "Local", groups[main_name]))
    return local_list

def _get_latest_commit_via_atom(slug, branch="main"):
    for br in (branch, "master" if branch == "main" else "main"):
        feed_url = f"https://{HOST_WEB}/{slug}/commits/{br}.atom"
        try:
            r = requests.get(feed_url, timeout=8, headers=HEADERS)
            if r.status_code == 200:
                m_sha = re.search(r"Commit/([0-9a-f]{40})", r.text)
                m_date = re.search(r"<updated>([^<]+)</updated>", r.text)
                if m_sha:
                    return m_sha.group(1), (m_date.group(1) if m_date else ""), br
        except Exception:
            pass
    return "", "", branch


def _fetch_repo_folders(slug, branch):
    api_url = f"https://{HOST_API}/repos/{slug}/contents"
    try:
        r = requests.get(api_url, timeout=10, headers=HEADERS)
        if r.status_code == 200:
            return [it["name"] for it in r.json() if it.get("type") == "dir" and not it["name"].startswith(".")]
        debug_log("GITHUB", f"[{slug}] API devolviÃ³ HTTP {r.status_code}, usando respaldo web sin lÃ­mite...")
    except Exception as e:
        debug_log("GITHUB", f"[{slug}] Error API ({e}), usando respaldo web...")

    web_url = f"https://{HOST_WEB}/{slug}"
    r = requests.get(web_url, timeout=12, headers=HEADERS)
    r.raise_for_status()
    pattern = rf'href="/{re.escape(slug)}/tree/{re.escape(branch)}/([^"/]+)"'
    found = re.findall(pattern, r.text)
    if not found:
        other_br = "master" if branch == "main" else "main"
        pattern2 = rf'href="/{re.escape(slug)}/tree/{other_br}/([^"/]+)"'
        found = re.findall(pattern2, r.text)
    return sorted(set(name for name in found if not name.startswith(".")))


def _fetch_single_toc(slug, branches, main_name):

        try:

            if tr.status_code == 200:

        except Exception:
            pass
    return "Desconocida", branches[0]


def check_github_addons(wow_dir, repo_urls, force_refresh=False):

            try:
                folders = _fetch_repo_folders(slug, detected_branch)
                groups = group_addon_folders(folders)
                branches = [detected_branch, "master" if detected_branch == "main" else "main"]

                    }

                        if remote_ver != "Desconocida":
                            detected_branch = used_branch
                        repo_entries.append({"name": main_name, "remote_ver": remote_ver, "modules": modules})

                cache[repo_url] = {
                    "sha": latest_sha,
                    "date": latest_date,
                    "branch": detected_branch,
                    "addons": repo_entries,
                }
                cache_dirty = True
                debug_log("CACHE", f"[{slug}] Guardados {len(repo_entries)} addons en cachÃ©.")
            except Exception:
                debug_log("GITHUB_ERR", f"Error en {repo_url}:\n{traceback.format_exc()}")
                repo_entries = cached_repo.get("addons", [])

        for entry in repo_entries:
            main_name = entry["name"]
            if main_name in seen_mains:

"""Read the active WoW client's realmlist without modifying it."""

import ipaddress
from pathlib import Path
import re
import shlex


DEFAULT_HOST = "wow.reinosdeleyenda.org"
DEFAULT_PORT = 3724


def validate_endpoint(host, port):
    host = host.strip().lower().rstrip(".")
    if host.startswith("[") and host.endswith("]"):
        host = host[1:-1]
    if ":" in host:
        try:
            host = str(ipaddress.IPv6Address(host))
        except ValueError:
            raise ValueError("Introduce una direcciÃ³n de servidor vÃ¡lida.") from None
    elif not re.fullmatch(r"(?=.{1,253}\Z)[a-z0-9](?:[a-z0-9-]{0,61}[a-z0-9])?"
                          r"(?:\.[a-z0-9](?:[a-z0-9-]{0,61}[a-z0-9])?)*", host):
        raise ValueError("Introduce una direcciÃ³n de servidor vÃ¡lida.")
    if not re.fullmatch(r"[0-9]{1,5}", str(port).strip()) or not 1 <= int(port) <= 65535:
        raise ValueError("El puerto debe estar entre 1 y 65535.")
    return host, int(port)


def read_realmlist(path):
    """Return a valid host and port or the project's default."""
    try:
        lines = Path(path).read_text(encoding="utf-8-sig", errors="replace").splitlines()
    except (OSError, TypeError):
        lines = ()
    for line in lines:
        if not re.match(r"^\s*set\s+realmlist\s+", line, re.IGNORECASE):
            continue
        try:
            _, _, address, *_ = shlex.split(line, comments=True)
            if address.startswith("["):
                tail = re.fullmatch(r"\[([^\]]+)\](?::([0-9]+))?", address)
                if not tail:
                    continue
                host, port = tail.group(1), tail.group(2) or DEFAULT_PORT
            elif address.count(":") == 1:
                host, port = address.split(":")
            else:
                host, port = address, DEFAULT_PORT
            return validate_endpoint(host, port)
        except ValueError:
            continue
    return DEFAULT_HOST, DEFAULT_PORT

class ChatSession(GameUI):
    """A single, explicitly started chat connection."""

    def __init__(self, host, port, username, password):
        self.events = queue.Queue()
        self.actions = queue.Queue()
        self.stopping = threading.Event()
        self.game = Game(lambda: self, host, port, username, password, LogLevel.Info)
        self.thread = threading.Thread(target=self._run, name="launcher-chat", daemon=True)
        self.realms = ()
        self.characters = ()
        self.phase = "auth"
        self.deadline = None

    def start(self):
        self.thread.start()

    def _run(self):
        try:
            self.game.start()
        except Exception:
            # Do not send exception/connection objects containing credentials to Tk or a log.
            self._emit("error", "No se pudo completar la sesiÃ³n de chat.")
        finally:
            self.events.put(("closed", None))

    def stop(self):
        self.stopping.set()
        self.game.cancel()

    def choose_realm(self, index):
        self.actions.put(("realm", index))

    def choose_character(self, index):
        self.actions.put(("character", index))

    def send(self, mode, message, target=""):
        message, target = message.strip(), target.strip()
        if mode not in SEND_MODES:
            raise ValueError("Selecciona un tipo de mensaje.")
        if not message or len(message.encode("utf-8")) > 255 or any(c in message for c in "\r\n\0"):
            raise ValueError("Escribe un mensaje de una lÃ­nea (mÃ¡ximo 255 bytes).")
        if mode == "Susurro" and (not target or len(target.encode("utf-8")) > 64
                                  or any(c.isspace() or c == "\0" for c in target)):
            raise ValueError("Escribe un nombre de destinatario vÃ¡lido.")
        self.actions.put(("send", (mode, message, target)))

    def update(self):
        if self.deadline is not None and monotonic() >= self.deadline:
            self._emit("error", "El servidor no respondiÃ³ a tiempo. Vuelve a conectar.")
            self.game.exit()
            return
        try:
            kind, payload = self.actions.get(timeout=.1)
        except queue.Empty:
            return
        if self.stopping.is_set():
            return
        try:
            self._handle_action(kind, payload)
        except OSError:
            self._emit("error", "Se perdiÃ³ la conexiÃ³n con el servidor.")
            self.game.exit()

    def _handle_action(self, kind, payload):
        if kind == "realm" and self.phase == "realm" and 0 <= payload < len(self.realms):
            self.phase = "world"
            self._emit("status", f"Conectando con {self.realms[payload].name}â€¦")
            self.game.connect_to(self.realms[payload])
            self.deadline = monotonic() + 30
        elif kind == "characters" and self.phase == "world":
            self.deadline = None
            if payload:
                self.characters = payload
                self.phase = "character"
                self._emit("characters", [(c.name, c.level) for c in payload])
            else:
                self._emit("error", "No hay personajes en este reino.")
                self.game.exit()
        elif kind == "character" and self.phase == "character" and 0 <= payload < len(self.characters):
            char = self.characters[payload]
            self.phase = "entering"
            self.deadline = monotonic() + 30
            self.game.world.selected_character = char
            self._emit("status", f"Entrando con {char.name}â€¦")
            packet = OutPacket(WorldCommand.CMSG_PLAYER_LOGIN)
            packet.write_u64(char.guid)
            self.game.send_packet(packet)
        elif kind == "ready" and self.phase == "entering":
            self.deadline = None
            self.phase = "ready"
            self._emit("ready", self.game.world.selected_character.name)
        elif kind == "login_failed" and self.phase == "entering":
            self._emit("error", f"El servidor rechazÃ³ la entrada del personaje (cÃ³digo {payload}).")
            self.game.exit()
        elif kind == "send" and self.phase == "ready":
            mode, body, target = payload
            char = self.game.world.selected_character
            packet = OutPacket(WorldCommand.CMSG_MESSAGECHAT)
            packet.write_u32(SEND_MODES[mode])
            packet.write_u32(Language.Orcish if is_horde(char.race) else Language.Common)
            if mode == "Susurro":
                packet.write_cstring(target)
            packet.write_cstring(body)
            self.game.send_packet(packet)
        elif kind == "send":
            self._emit("error", "El personaje todavÃ­a no ha entrado en el mundo.")

    def _emit(self, kind, payload):
        if not self.stopping.is_set():
            self.events.put((kind, payload))

    def present_realm_list(self, realm_list):

import base64
import ctypes
from ctypes import wintypes

crypt32 = ctypes.windll.crypt32
kernel32 = ctypes.windll.kernel32


class DATA_BLOB(ctypes.Structure):
    _fields_ = [
        ("cbData", wintypes.DWORD),
        ("pbData", ctypes.POINTER(ctypes.c_byte)),
    ]


def encrypt_password(plain_text):
    if not plain_text:
        return ""
    raw_bytes = plain_text.encode("utf-8")
    in_blob = DATA_BLOB(len(raw_bytes), ctypes.cast(ctypes.create_string_buffer(raw_bytes), ctypes.POINTER(ctypes.c_byte)))
    out_blob = DATA_BLOB()

    # CRYPTPROTECT_UI_FORBIDDEN = 0x01
    if not crypt32.CryptProtectData(ctypes.byref(in_blob), None, None, None, None, 0x01, ctypes.byref(out_blob)):
        raise ctypes.WinError()

    try:
        encrypted = ctypes.string_at(out_blob.pbData, out_blob.cbData)
        return "DPAPI:" + base64.b64encode(encrypted).decode("ascii")
    finally:
        kernel32.LocalFree(out_blob.pbData)


def decrypt_password(stored_value):
    if not stored_value:
        return ""
    # Retrocompatibilidad: si hubiera una clave antigua en texto plano, la devuelve tal cual
    if not stored_value.startswith("DPAPI:"):
        return stored_value

    encrypted = base64.b64decode(stored_value[6:])
    in_blob = DATA_BLOB(len(encrypted), ctypes.cast(ctypes.create_string_buffer(encrypted), ctypes.POINTER(ctypes.c_byte)))
    out_blob = DATA_BLOB()

    if not crypt32.CryptUnprotectData(ctypes.byref(in_blob), None, None, None, None, 0x01, ctypes.byref(out_blob)):
        raise ctypes.WinError()

    try:
        decrypted = ctypes.string_at(out_blob.pbData, out_blob.cbData)
        return decrypted.decode("utf-8")
    finally:
        kernel32.LocalFree(out_blob.pbData)

VK_RETURN = 0x0D
VK_SHIFT = 0x10
KEYEVENTF_KEYUP = 0x0002


def _send_password_physical(password):
    user32 = ctypes.windll.user32
    for ch in password:
        vk_scan = user32.VkKeyScanW(ord(ch))
        if vk_scan == -1:
            continue
        vk = vk_scan & 0xFF
        shift_state = (vk_scan >> 8) & 0xFF

        if shift_state & 1:
            user32.keybd_event(VK_SHIFT, 0, 0, 0)
            time.sleep(0.01)

        user32.keybd_event(vk, 0, 0, 0)
        time.sleep(0.015)
        user32.keybd_event(vk, 0, KEYEVENTF_KEYUP, 0)

        if shift_state & 1:
            time.sleep(0.01)
            user32.keybd_event(VK_SHIFT, 0, KEYEVENTF_KEYUP, 0)

        time.sleep(0.02)

    time.sleep(0.15)
    user32.keybd_event(VK_RETURN, 0, 0, 0)
    time.sleep(0.05)
    user32.keybd_event(VK_RETURN, 0, KEYEVENTF_KEYUP, 0)

def _find_wow_hwnd(pid):
    user32 = ctypes.windll.user32
    found_by_pid = []
    found_by_title = []

    @ctypes.WINFUNCTYPE(wintypes.BOOL, wintypes.HWND, wintypes.LPARAM)
    def enum_cb(hwnd, _):
        if not user32.IsWindowVisible(hwnd):
            return True
        rect = wintypes.RECT()
        user32.GetClientRect(hwnd, ctypes.byref(rect))
        if rect.right <= 400 or rect.bottom <= 300:
            return True

        proc_id = ctypes.c_ulong()
        user32.GetWindowThreadProcessId(hwnd, ctypes.byref(proc_id))
        if proc_id.value == pid:
            found_by_pid.append(hwnd)

        buf = ctypes.create_unicode_buffer(128)
        user32.GetWindowTextW(hwnd, buf, 128)
        if buf.value == "World of Warcraft":
            found_by_title.append(hwnd)
        return True

    user32.EnumWindows(enum_cb, 0)
    if found_by_pid:
        return found_by_pid[0]
    if found_by_title:
        return found_by_title[0]
    return None

def load_local_logo(max_size=(380, 220)):
    debug_log("LOGO", f"Cargando logo local desde: {LOCAL_LOGO_FILE}")
    if not os.path.exists(LOCAL_LOGO_FILE):
        raise FileNotFoundError(f"No se encontrÃ³ logo.png en:\n{LOCAL_LOGO_FILE}")
    img = Image.open(LOCAL_LOGO_FILE).convert("RGBA")
    img.thumbnail(max_size, Image.Resampling.LANCZOS)
    return ImageTk.PhotoImage(img)


def fetch_latest_news():
    articles = []
    debug_log("NEWS", f"Consultando API WP: {WP_API_URL}")

    try:
        r = requests.get(WP_API_URL, timeout=10, headers=HEADERS)
        debug_log("NEWS", f"WP API Status: {r.status_code}")
        if r.status_code == 200:
            for post in r.json():
                title = BeautifulSoup(post.get("title", {}).get("rendered", ""), "html.parser").get_text(strip=True)
                excerpt = BeautifulSoup(post.get("excerpt", {}).get("rendered", ""), "html.parser").get_text(" ", strip=True)
                date_str = post.get("date", "")[:10]
                link = post.get("link", NEWS_URL)
                if title:
                    articles.append((date_str, title, excerpt, link))
    except Exception:
        debug_log("NEWS_ERR", traceback.format_exc())

    if not articles:
        debug_log("NEWS", f"Probando scraping HTML en: {NEWS_URL}")
        try:
            r = requests.get(NEWS_URL, timeout=10, headers=HEADERS)
            debug_log("NEWS", f"HTML Status: {r.status_code}")
            soup = BeautifulSoup(r.text, "html.parser")
            for art in soup.find_all("article")[:8]:
                h2 = art.find(["h2", "h1"])
                if not h2:
                    continue
                title = h2.get_text(strip=True)
                a_tag = h2.find("a")
                link = a_tag["href"] if a_tag and a_tag.has_attr("href") else NEWS_URL
                time_tag = art.find("time")
                date_str = time_tag.get_text(strip=True) if time_tag else ""
                p_tag = art.find("p")
                excerpt = p_tag.get_text(" ", strip=True) if p_tag else ""
                articles.append((date_str, title, excerpt, link))
        except Exception as e:
            debug_log("NEWS_ERR", traceback.format_exc())
            articles.append(("Error", "No se pudieron cargar las noticias", str(e), NEWS_URL))

    debug_log("NEWS", f"Noticias cargadas: {len(articles)}")
    return articles

    draw = ImageDraw.Draw(img)

class TrayManager:
    def __init__(self, on_show_cb, on_play_cb, on_quit_cb):
        self.on_show_cb = on_show_cb
        self.on_play_cb = on_play_cb
        self.on_quit_cb = on_quit_cb
        self.icon = None

    def start(self):
        if self.icon is not None:
            return
        menu = pystray.Menu(
            pystray.MenuItem("Abrir Launcher", lambda icon, item: self.on_show_cb(), default=True),
            pystray.MenuItem("Jugar WoW", lambda icon, item: self.on_play_cb()),
            pystray.Menu.SEPARATOR,
            pystray.MenuItem("Salir", lambda icon, item: self.on_quit_cb()),
        )
        self.icon = pystray.Icon("RDL_Launcher", _build_tray_icon_image(), "Reinos de Leyenda Launcher", menu)
        threading.Thread(target=self.icon.run, daemon=True).start()
        debug_log("TRAY", "Icono de bandeja del sistema iniciado.")

    def stop(self):
        if self.icon is not None:
            try:
                self.icon.stop()
            except Exception:
                pass
            self.icon = None

import os
import re
import subprocess
import sys
import requests
from config import BASE_DIR, HOST_API, HOST_WEB, LAUNCHER_RELEASES_URL, LAUNCHER_REPO, LAUNCHER_VERSION, debug_log

HEADERS = {"User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) RDL-Launcher/1.0"}


def _parse_ver_tuple(ver_str):
    nums = [int(x) for x in re.findall(r"\d+", ver_str or "")]
    return tuple(nums) if nums else (0,)


def is_newer_version(remote_ver, local_ver=LAUNCHER_VERSION):
    return _parse_ver_tuple(remote_ver) > _parse_ver_tuple(local_ver)


def check_launcher_update():
    """Devuelve (has_update, remote_ver, download_url) usando releases.atom sin lÃ­mites de API."""
    remote_ver = ""
    raw_tag = ""
    download_url = LAUNCHER_RELEASES_URL

    # 1. Feed Atom de Releases (sin rate-limit de API y funciona al instante tras publicar)
    atom_url = f"https://{HOST_WEB}/{LAUNCHER_REPO}/releases.atom"
    try:
        r = requests.get(atom_url, timeout=8, headers=HEADERS)
        debug_log("UPDATER", f"Atom status: {r.status_code}")
        if r.status_code == 200:
            m = re.search(rf'/{re.escape(LAUNCHER_REPO)}/releases/tag/([^"\'<>\s]+)', r.text)
            if m:
                raw_tag = m.group(1)
                remote_ver = raw_tag.lstrip("vV")
                download_url = f"https://{HOST_WEB}/{LAUNCHER_REPO}/releases/download/{raw_tag}/RDL_Launcher.exe"
    except Exception as e:
        debug_log("UPDATER", f"Error leyendo releases.atom: {e}")

    # 2. Respaldo: API de GitHub Releases (por si el .exe tiene otro nombre)
    if not remote_ver:
        api_url = f"https://{HOST_API}/repos/{LAUNCHER_REPO}/releases"
        try:
            r = requests.get(api_url, timeout=8, headers=HEADERS)
            if r.status_code == 200 and r.json():
                latest_rel = r.json()[0]
                raw_tag = latest_rel.get("tag_name", "")
                remote_ver = raw_tag.lstrip("vV")
                for asset in latest_rel.get("assets", []):
                    if asset.get("name", "").lower().endswith(".exe"):
                        download_url = asset.get("browser_download_url", download_url)
                        break
        except Exception as e:
            debug_log("UPDATER", f"Error en API releases: {e}")

    debug_log("UPDATER", f"VersiÃ³n local: {LAUNCHER_VERSION} | Remota: {remote_ver} | URL: {download_url}")
    return is_newer_version(remote_ver, LAUNCHER_VERSION), remote_ver, download_url


def download_and_apply_exe_update(download_url, progress_cb=None):
    """Descarga el nuevo .exe y lanza un script .bat desvinculado para reemplazarlo y reiniciar."""
    if not getattr(sys, "frozen", False):
        return False

    current_exe = sys.executable
    old_exe = current_exe + ".old"
    new_exe = os.path.join(BASE_DIR, "RDL_Launcher_new.exe")
    bat_path = os.path.join(BASE_DIR, "_update_launcher.bat")

    if os.path.exists(old_exe):
        try:
            os.remove(old_exe)
        except Exception:
            pass

    debug_log("UPDATER", f"Descargando actualizaciÃ³n desde: {download_url}")
    with requests.get(download_url, stream=True, timeout=60, headers=HEADERS, allow_redirects=True) as r:
        if r.status_code == 404:
            debug_log("UPDATER", "Asset RDL_Launcher.exe no encontrado directamente (404), abriendo web.")
            return False
        r.raise_for_status()
        total = int(r.headers.get("content-length", 0))
        downloaded = 0
        with open(new_exe, "wb") as f:
            for chunk in r.iter_content(chunk_size=65536):
                if chunk:
                    f.write(chunk)
                    downloaded += len(chunk)
                    if progress_cb and total > 0:
                        progress_cb(int(downloaded * 100 / total))

    # Limpia _MEIPASS2, renombra el exe antiguo y arranca limpio vÃ­a explorer.exe
    bat_script = f"""@echo off
set _MEIPASS2=
timeout /t 2 /nobreak > nul
:retry
move /y "{current_exe}" "{old_exe}" > nul 2>&1
if exist "{current_exe}" (
    timeout /t 1 /nobreak > nul
    goto retry
)
move /y "{new_exe}" "{current_exe}" > nul
explorer.exe "{current_exe}"
timeout /t 3 /nobreak > nul
del /f /q "{old_exe}" > nul 2>&1
del "%~f0"
"""
    with open(bat_path, "w", encoding="utf-8") as f:
        f.write(bat_script)

    subprocess.Popen(
        ["cmd", "/c", bat_path],
        creationflags=subprocess.CREATE_NO_WINDOW | subprocess.DETACHED_PROCESS,
        close_fds=True,
    )
    return True
